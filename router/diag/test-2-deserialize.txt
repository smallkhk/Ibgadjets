# TEST 2 — fetch AND parse the JSON. Nothing else.
# Only run this if TEST 1 survived.
# If the router reboots here, :deserialize is the culprit.

:local ibgUrl "https://ibphone.eclipselivecam.online/api/router-sync.php"
:local ibgKey "21e1e9a7a22d47e43e9fd75d52b998821336d3c99e653d3747cb57069e6aab55"

:do {
    :local res [/tool fetch url=$ibgUrl \
        http-method=get \
        http-header-field=("X-Sync-Key: " . $ibgKey) \
        output=user as-value]
    :local d ($res->"data")
    :if ([:pick $d 0 1] != "{") do={ :error "notjson" }

    :local doc [:deserialize from=json value=$d]
    :log info "ibg-test2: deserialize SURVIVED"
    :log info ("ibg-test2: ok=" . ($doc->"ok"))
    :local users ($doc->"users")
    :log info ("ibg-test2: users=" . [:len $users])
    :foreach u in=$users do={
        :log info ("ibg-test2: user " . ($u->"username") . " mb=" . ($u->"data_mb"))
    }
    :log info "ibg-test2: SURVIVED ALL"
} on-error={
    :log warning "ibg-test2: error (but no crash)"
}
